#include <Wire.h>
#include <WiFi.h>
#include <WebServer.h>
#include <Adafruit_PWMServoDriver.h>

Adafruit_PWMServoDriver pwm(0x40);
WebServer server(80);

const char* AP_SSID = "RobotDog";
const char* AP_PASS = "12345678";   // 至少 8 碼

// ---------- 設定 ----------
constexpr uint16_t SERVO_MIN_US = 600;
constexpr uint16_t SERVO_MAX_US = 2300;
constexpr uint16_t HYST_US      = 5;

const uint8_t legs[4][3] = {
  {0, 1, 2},
  {4, 5, 6},
  {8, 9, 10},
  {12, 13, 14},
};
const float HOME[4][3] = {
  {95, 160, 160},
  {90, 20, 20},
  {90, 20, 22},
  {95, 160, 160}
};

float    curAngle[16];
uint16_t lastUs[16] = {0};

// ---------- 底層輸出（含遲滯）----------
void setServoEx(uint8_t ch, float angle, bool force) {
  angle = constrain(angle, 0.0f, 180.0f);
  curAngle[ch] = angle;

  uint16_t us = SERVO_MIN_US + (uint16_t)(angle * (SERVO_MAX_US - SERVO_MIN_US) / 180.0f + 0.5f);

  if (!force && lastUs[ch] != 0 && abs((int)us - (int)lastUs[ch]) < HYST_US) return;

  lastUs[ch] = us;
  pwm.writeMicroseconds(ch, us);
}

// ---------- 腳緩動 ----------
constexpr uint32_t UPDATE_MS = 20;

struct Pose { float a[3]; uint16_t ms; };
struct Seq  { const Pose* p; uint8_t n; };
struct Phase { Seq s[4]; };   // A B C D 各自要跑的動作，p = nullptr 代表不動

#define COUNT(x) ((uint8_t)(sizeof(x) / sizeof((x)[0])))
#define NONE {nullptr, 0}

// ===== A、B 腿 =====
// 第一輪（從 reload 出發）
const Pose WALK_A[] = { {{ 95,160,160},400}, {{ 30,130, 90},400}, {{ 30,130,110},400} };
const Pose WALK_B[] = { {{ 90, 20, 20},400}, {{150, 45, 90},400}, {{150, 45, 65},400} };
// 循環輪（起點 = 上一輪身體階段結尾 A(95,130,110) B(90,45,65)）
const Pose LOOP_A[] = { {{ 95,130, 90},400}, {{ 30,130, 90},400}, {{ 30,130,110},400} };
const Pose LOOP_B[] = { {{ 90, 45, 90},400}, {{150, 45, 90},400}, {{150, 45, 65},400} };

// ===== 身體往前（A 95、B 90、C 48、D 135）=====
// 第一輪：C、D 只掃髖，femur/tibia 維持 HOME
const Pose BODY_A[] = { {{ 95,130,110},400} };
const Pose BODY_B[] = { {{ 90, 45, 65},400} };
const Pose BODY_C[] = { {{ 135, 20, 22},400} };
const Pose BODY_D[] = { {{ 45,160,160},400} };
// 循環輪：C、D 掃髖的同時直接站立（省掉階段 4、5）
const Pose LBODY_C[] = { {{135, 55, 75},400} };
const Pose LBODY_D[] = { {{45,130,130},400} };

// ===== C、D 站立（只有第一輪）=====
const Pose STAND_C[] = { {{ 135, 20, 22},400}, {{ 135, 55, 90},400}, {{ 135, 55, 75},400} };
const Pose STAND_D[] = { {{45,160,160},400}, {{45,130, 90},400}, {{45,130,130},400} };

// ===== C、D 往前 =====
const Pose FWD_C[] = { {{ 135, 55, 75},400}, {{ 135, 55, 90},400}, {{ 90, 55, 75},400} };
const Pose FWD_D[] = { {{45,130,130},400}, {{45,130, 90},400}, {{ 95,130,130},400} };

constexpr uint8_t FIRST_PHASES = 7;
constexpr uint8_t LOOP_PHASES  = 5;

const Phase FIRST_SET[FIRST_PHASES] = {
  {{ {WALK_A, COUNT(WALK_A)}, NONE, NONE, NONE }},                      // 1 A 腿
  {{ NONE, {WALK_B, COUNT(WALK_B)}, NONE, NONE }},                      // 2 B 腿
  {{ {BODY_A,1}, {BODY_B,1}, {BODY_C,1}, {BODY_D,1} }},                 // 3 身體往前
  {{ NONE, NONE, {STAND_C, COUNT(STAND_C)}, NONE }},                    // 4 C 站立
  {{ NONE, NONE, NONE, {STAND_D, COUNT(STAND_D)} }},                    // 5 D 站立
  {{ NONE, NONE, {FWD_C, COUNT(FWD_C)}, NONE }},                        // 6 C 往前
  {{ NONE, NONE, NONE, {FWD_D, COUNT(FWD_D)} }},                        // 7 D 往前
};

const Phase LOOP_SET[LOOP_PHASES] = {
  {{ {LOOP_A, COUNT(LOOP_A)}, NONE, NONE, NONE }},                      // 1 A 腿
  {{ NONE, {LOOP_B, COUNT(LOOP_B)}, NONE, NONE }},                      // 2 B 腿
  {{ {BODY_A,1}, {BODY_B,1}, {LBODY_C,1}, {LBODY_D,1} }},               // 3 身體往前（C、D 同時站立）
  {{ NONE, NONE, {FWD_C, COUNT(FWD_C)}, NONE }},                        // 4 C 往前
  {{ NONE, NONE, NONE, {FWD_D, COUNT(FWD_D)} }},                        // 5 D 往前
};

const Phase* curSet    = FIRST_SET;
uint8_t      curPhases = FIRST_PHASES;

struct LegMotion {
  const Pose* seq = nullptr;
  uint8_t  len = 0, idx = 0, leg = 0;
  uint32_t t0 = 0;
  float    from[3] = {0, 0, 0};
  bool     active = false;

  void beginStep() {
    t0 = millis();
    for (int i = 0; i < 3; i++) from[i] = curAngle[legs[leg][i]];
  }

  void start(const Pose* s, uint8_t n) {
    for (int i = 0; i < 3; i++) {
      if (isnan(curAngle[legs[leg][i]])) {
        Serial.println("位置未知，請先執行 r (reload)");
        return;
      }
    }
    seq = s; len = n; idx = 0; active = true;
    beginStep();
  }

  void update() {
    if (!active) return;

    const Pose& p = seq[idx];
    float u = p.ms ? (millis() - t0) / (float)p.ms : 1.0f;
    if (u > 1.0f) u = 1.0f;
    float e = u * u * (3.0f - 2.0f * u);   // smoothstep
    bool done = (u >= 1.0f);

    for (int i = 0; i < 3; i++)
      setServoEx(legs[leg][i], from[i] + (p.a[i] - from[i]) * e, done);

    if (done) {
      if (++idx >= len) active = false;
      else beginStep();
    }
  }
};

LegMotion motionLeg[4];
uint8_t phase   = 0;       // 0 閒置，1~7 = 目前階段
bool    looping = false;

// ---------- 指令 ----------
void stopAll() {
  Serial.println("clearing");
  phase = 0;
  looping = false;
  curSet = FIRST_SET; curPhases = FIRST_PHASES;
  for (int i = 0; i < 4; i++) motionLeg[i].active = false;
  for (int ch = 0; ch < 16; ch++) {
    pwm.setPWM(ch, 0, 4096);
    curAngle[ch] = NAN;
    lastUs[ch] = 0;
  }
  Serial.println("cleared");
}

void reload() {
  Serial.println("reloading");
  phase = 0;
  looping = false;
  curSet = FIRST_SET; curPhases = FIRST_PHASES;
  for (int i = 0; i < 4; i++) motionLeg[i].active = false;
  for (int leg = 0; leg < 3; leg += 2) {
    for (int j = 0; j < 3; j++) {
      setServoEx(legs[leg][j],     HOME[leg][j],     true);
      setServoEx(legs[leg + 1][j], HOME[leg + 1][j], true);
      delay(j < 2 ? 500 : 2000);
    }
  }
  Serial.println("reload done");
}

// ---------- 階段控制 ----------
void startPhase(uint8_t p) {
  phase = p;
  Serial.print("階段 "); Serial.print(p); Serial.print("/"); Serial.println(curPhases);
  const Phase& ph = curSet[p - 1];
  for (uint8_t i = 0; i < 4; i++)
    if (ph.s[i].p) motionLeg[i].start(ph.s[i].p, ph.s[i].n);
  for (uint8_t i = 0; i < 4; i++)
    if (ph.s[i].p && !motionLeg[i].active) { phase = 0; return; }   // 位置未知等失敗
}

void updatePhase() {
  if (phase == 0) return;
  for (uint8_t i = 0; i < 4; i++)
    if (motionLeg[i].active) return;   // 本階段還有腳在動

  if (phase < curPhases) {
    startPhase(phase + 1);
  } else if (looping) {
    Serial.println("下一輪（循環組）");
    curSet = LOOP_SET; curPhases = LOOP_PHASES;
    startPhase(1);
  } else {
    Serial.println("全部完成");
    phase = 0;
  }
}

void startWalkLoop() {
  looping = true;
  curSet = FIRST_SET; curPhases = FIRST_PHASES;
  startPhase(1);
}

void action3() {
  Serial.println("動作 3：尚未設定");
}

// ---------- 指令入口（序列埠與網頁共用）----------
void handleCommand(char c) {
  switch (c) {
    case 's': stopAll();       break;
    case 'r': reload();        break;
    case 'w': startWalkLoop(); break;
    case 'e': looping = false; Serial.println("本輪結束後停止"); break;
    case 'c': action3();       break;
    default:  Serial.println("指令：s 斷電 / r 歸位 / w 循環走 / e 本輪後停止 / c"); break;
  }
}

void handleRoot() { server.send_P(200, "text/html; charset=utf-8", PAGE); }

void handleCmd() {
  if (!server.hasArg("c") || server.arg("c").length() == 0) {
    server.send(400, "text/plain; charset=utf-8", "missing c");
    return;
  }
  char c = server.arg("c")[0];
  handleCommand(c);
  server.send(200, "text/plain; charset=utf-8", String("OK: ") + c);
}

// ---------- Arduino ----------
void setup() {
  Serial.begin(115200);
  Wire.begin(8, 9);

  if (!pwm.begin()) {
    Serial.println("PCA9685 Not Found.");
    while (true) delay(1000);
  } else {
    Serial.println("PCA9685 Found.");
  }
  pwm.setOscillatorFrequency(27000000);
  pwm.setPWMFreq(50);

  for (uint8_t i = 0; i < 4; i++) motionLeg[i].leg = i;

  for (int ch = 0; ch < 16; ch++) {
    pwm.setPWM(ch, 0, 4096);
    curAngle[ch] = NAN;
    lastUs[ch] = 0;
  }
  Serial.println("Stop all output");

  WiFi.mode(WIFI_AP);
  WiFi.softAP(AP_SSID, AP_PASS);
  Serial.print("AP IP: "); Serial.println(WiFi.softAPIP());

  server.on("/", handleRoot);
  server.on("/cmd", handleCmd);
  server.begin();
}

void loop() {
  server.handleClient();

  if (Serial.available() > 0) {
    char c = Serial.read();
    if (c != '\n' && c != '\r') handleCommand(c);
  }

  static uint32_t last = 0;
  uint32_t now = millis();
  if (now - last >= UPDATE_MS) {
    last = now;
    for (uint8_t i = 0; i < 4; i++) motionLeg[i].update();
    updatePhase();
  }
}
