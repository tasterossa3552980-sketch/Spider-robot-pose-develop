#include <Wire.h>
#include <WiFi.h>
#include <WebServer.h>
#include <Adafruit_PWMServoDriver.h>

Adafruit_PWMServoDriver pwm(0x40);
WebServer server(80);

const char* AP_SSID = "RobotDog";
const char* AP_PASS = "12345678";   // 至少 8 碼

// ---------- 設定 ----------
constexpr uint16_t SERVO_MIN_US = 600;
constexpr uint16_t SERVO_MAX_US = 2300;
constexpr uint16_t HYST_US      = 5;

const uint8_t legs[4][3] = {
  {0, 1, 2},
  {4, 5, 6},
  {8, 9, 10},
  {12, 13, 14},
};
const float HOME[4][3] = {
  {95, 160, 160},
  {90, 20, 20},
  {90, 20, 22},
  {95, 160, 160}
};

float    curAngle[16];
uint16_t lastUs[16] = {0};

// ---------- 步態參數（A 左前、B 右前、C 左後、D 右後）----------
constexpr float S = 30;                                    // 髖行程 ±S（中間側上限 30/150，勿超過）
const float   CENTER[4]   = {95, 90, 90, 95};              // 髖中立
const int8_t  DIR[4]      = {-1, +1, -1, +1};              // 往前時髖角度的變化方向
const float   STAND[4][2] = {{130,110},{45,65},{55,75},{130,130}};  // 站姿 femur, tibia
const float   LIFT[4]     = {-25, +25, +25, -25};          // 抬腿時 tibia 的變化量
const uint8_t ORDER[4]    = {0, 3, 1, 2};                  // 抬腿順序 A D B C
constexpr uint16_t READY_MS = 800;
constexpr uint16_t LIFT_MS  = 200;                         // 抬腿 / 放腿
constexpr uint16_t SWING_MS = 400;                         // 擺動（同時身體前進）

// ---------- 底層輸出（含遲滯）----------
void setServoEx(uint8_t ch, float angle, bool force) {
  angle = constrain(angle, 0.0f, 180.0f);
  curAngle[ch] = angle;

  uint16_t us = SERVO_MIN_US + (uint16_t)(angle * (SERVO_MAX_US - SERVO_MIN_US) / 180.0f + 0.5f);

  if (!force && lastUs[ch] != 0 && abs((int)us - (int)lastUs[ch]) < HYST_US) return;

  lastUs[ch] = us;
  pwm.writeMicroseconds(ch, us);
}

// ---------- 腳緩動 ----------
constexpr uint32_t UPDATE_MS = 20;

struct Pose  { float a[3]; uint16_t ms; };
struct Seq   { const Pose* p; uint8_t n; };
struct Phase { Seq s[4]; };

void setPose(Pose& p, uint8_t l, float x, float tibiaOff, uint16_t ms);

constexpr uint8_t PHASES = 5;      // 1 = 就位，2~5 = 依序抬 A D B C
Pose  gaitPose[PHASES][4][3];
Phase gaitPhase[PHASES];

float hipAng(uint8_t l, float x) { return CENTER[l] + DIR[l] * x; }

void setPose(Pose& p, uint8_t l, float x, float tibiaOff, uint16_t ms) {
  p.a[0] = hipAng(l, x);
  p.a[1] = STAND[l][0];
  p.a[2] = STAND[l][1] + tibiaOff;
  p.ms = ms;
}

// x = 髖相對中立的「向前量」，-S = 最後，+S = 最前
void buildGait() {
  float x[4];
  x[ORDER[0]] = -S;        // 即將抬的腿在最後
  x[ORDER[1]] = -S / 3;
  x[ORDER[2]] =  S / 3;
  x[ORDER[3]] =  S;        // 剛落地的腿在最前

  // 階段 1：從 HOME 就位（先 femur/tibia 到站姿，再髖到起始位置）
  for (uint8_t l = 0; l < 4; l++) {
    Pose* r = gaitPose[0][l];
    r[0].a[0] = HOME[l][0]; r[0].a[1] = STAND[l][0]; r[0].a[2] = STAND[l][1]; r[0].ms = READY_MS;
    setPose(r[1], l, x[l], 0, READY_MS);
    gaitPhase[0].s[l] = {r, 2};
  }

  // 階段 2~5：每隻腳抬、擺、放
  for (uint8_t k = 0; k < 4; k++) {
    uint8_t s = ORDER[k];
    for (uint8_t l = 0; l < 4; l++)
      setPose(gaitPose[k + 1][l][0], l, x[l], (l == s) ? LIFT[l] : 0, LIFT_MS);   // 抬腿

    for (uint8_t l = 0; l < 4; l++) {                                            // 更新髖位置
      if (l == s) x[l] = S;
      else        x[l] -= 2 * S / 3;
    }

    for (uint8_t l = 0; l < 4; l++) {
      Pose* q = gaitPose[k + 1][l];
      setPose(q[1], l, x[l], (l == s) ? LIFT[l] : 0, SWING_MS);                  // 擺動 / 撐地後掃
      setPose(q[2], l, x[l], 0, LIFT_MS);                                        // 放下
      gaitPhase[k + 1].s[l] = {q, 3};
    }
  }
}

struct LegMotion {
  const Pose* seq = nullptr;
  uint8_t  len = 0, idx = 0, leg = 0;
  uint32_t t0 = 0;
  float    from[3] = {0, 0, 0};
  bool     active = false;

  void beginStep() {
    t0 = millis();
    for (int i = 0; i < 3; i++) from[i] = curAngle[legs[leg][i]];
  }

  void start(const Pose* s, uint8_t n) {
    for (int i = 0; i < 3; i++) {
      if (isnan(curAngle[legs[leg][i]])) {
        Serial.println("位置未知，請先執行 r (reload)");
        return;
      }
    }
    seq = s; len = n; idx = 0; active = true;
    beginStep();
  }

  void update() {
    if (!active) return;

    const Pose& p = seq[idx];
    float u = p.ms ? (millis() - t0) / (float)p.ms : 1.0f;
    if (u > 1.0f) u = 1.0f;
    float e = u * u * (3.0f - 2.0f * u);   // smoothstep
    bool done = (u >= 1.0f);

    for (int i = 0; i < 3; i++)
      setServoEx(legs[leg][i], from[i] + (p.a[i] - from[i]) * e, done);

    if (done) {
      if (++idx >= len) active = false;
      else beginStep();
    }
  }
};

LegMotion motionLeg[4];
uint8_t phase   = 0;       // 0 閒置，1~5 = 目前階段
bool    looping = false;

// ---------- 指令 ----------
void stopAll() {
  Serial.println("clearing");
  phase = 0;
  looping = false;
  for (int i = 0; i < 4; i++) motionLeg[i].active = false;
  for (int ch = 0; ch < 16; ch++) {
    pwm.setPWM(ch, 0, 4096);
    curAngle[ch] = NAN;
    lastUs[ch] = 0;
  }
  Serial.println("cleared");
}

void reload() {
  Serial.println("reloading");
  phase = 0;
  looping = false;
  for (int i = 0; i < 4; i++) motionLeg[i].active = false;
  for (int leg = 0; leg < 3; leg += 2) {
    for (int j = 0; j < 3; j++) {
      setServoEx(legs[leg][j],     HOME[leg][j],     true);
      setServoEx(legs[leg + 1][j], HOME[leg + 1][j], true);
      delay(j < 2 ? 500 : 2000);
    }
  }
  Serial.println("reload done");
}

// ---------- 階段控制 ----------
void startPhase(uint8_t p) {
  phase = p;
  Serial.print("階段 "); Serial.print(p); Serial.print("/"); Serial.println(PHASES);
  const Phase& ph = gaitPhase[p - 1];
  for (uint8_t i = 0; i < 4; i++) motionLeg[i].start(ph.s[i].p, ph.s[i].n);
  for (uint8_t i = 0; i < 4; i++)
    if (!motionLeg[i].active) { phase = 0; return; }   // 位置未知等失敗
}

void updatePhase() {
  if (phase == 0) return;
  for (uint8_t i = 0; i < 4; i++)
    if (motionLeg[i].active) return;   // 本階段還有腳在動

  if (phase < PHASES) {
    startPhase(phase + 1);
  } else if (looping) {
    startPhase(2);                      // 循環：回到「抬 A」，不再重做就位
  } else {
    Serial.println("全部完成");
    phase = 0;
  }
}

void startWalkLoop() {
  looping = true;
  startPhase(1);
}

void action3() {
  Serial.println("動作 3：尚未設定");
}

// ---------- 指令入口（序列埠與網頁共用）----------
void handleCommand(char c) {
  switch (c) {
    case 's': stopAll();       break;
    case 'r': reload();        break;
    case 'w': startWalkLoop(); break;
    case 'e': looping = false; Serial.println("本輪結束後停止"); break;
    case 'c': action3();       break;
    default:  Serial.println("指令：s 斷電 / r 歸位 / w 循環走 / e 本輪後停止 / c"); break;
  }
}

void handleRoot() { server.send_P(200, "text/html; charset=utf-8", PAGE); }

void handleCmd() {
  if (!server.hasArg("c") || server.arg("c").length() == 0) {
    server.send(400, "text/plain; charset=utf-8", "missing c");
    return;
  }
  char c = server.arg("c")[0];
  handleCommand(c);
  server.send(200, "text/plain; charset=utf-8", String("OK: ") + c);
}

// ---------- Arduino ----------
void setup() {
  Serial.begin(115200);
  Wire.begin(8, 9);

  if (!pwm.begin()) {
    Serial.println("PCA9685 Not Found.");
    while (true) delay(1000);
  } else {
    Serial.println("PCA9685 Found.");
  }
  pwm.setOscillatorFrequency(27000000);
  pwm.setPWMFreq(50);

  for (uint8_t i = 0; i < 4; i++) motionLeg[i].leg = i;

  for (int ch = 0; ch < 16; ch++) {
    pwm.setPWM(ch, 0, 4096);
    curAngle[ch] = NAN;
    lastUs[ch] = 0;
  }
  Serial.println("Stop all output");

  buildGait();

  WiFi.mode(WIFI_AP);
  WiFi.softAP(AP_SSID, AP_PASS);
  Serial.print("AP IP: "); Serial.println(WiFi.softAPIP());

  server.on("/", handleRoot);
  server.on("/cmd", handleCmd);
  server.begin();
}

void loop() {
  server.handleClient();

  if (Serial.available() > 0) {
    char c = Serial.read();
    if (c != '\n' && c != '\r') handleCommand(c);
  }

  static uint32_t last = 0;
  uint32_t now = millis();
  if (now - last >= UPDATE_MS) {
    last = now;
    for (uint8_t i = 0; i < 4; i++) motionLeg[i].update();
    updatePhase();
  }
}
